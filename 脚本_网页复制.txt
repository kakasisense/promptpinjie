// ==UserScript==
// @name         网页内容复制
// @namespace    http://tampermonkey.net/
// @version      2026.09.01.2
// @description  网页悬浮复制工具，划词拾取文本、提取页面标题链接；面板带十字拖动按钮可自由拖拽，支持缩放，输入框聚焦自动上移避开输入法遮挡；一键复制内容框文本；悬浮球默认在页面左侧中间
// @author       @公众号：开关的txt
// @match        *://*/*
// @grant        GM_setClipboard
// ==/UserScript==

(function() {
    'use strict';
    const MAX_HISTORY = 10;
    const HISTORY_KEY = "copy_note_history";
    const INPUT_OFFSET = 220;

    // 悬浮球：默认左侧居中
    const floatBtn = document.createElement('div');
    floatBtn.style.position = 'fixed';
    floatBtn.style.left = '12px';
    floatBtn.style.top = '50%';
    floatBtn.style.transform = 'translateY(-50%)';
    floatBtn.style.zIndex = '999999';
    floatBtn.style.width = '44px';
    floatBtn.style.height = '44px';
    floatBtn.style.borderRadius = '50%';
    floatBtn.style.background = '#2277dd';
    floatBtn.style.color = '#fff';
    floatBtn.style.display = 'flex';
    floatBtn.style.alignItems = 'center';
    floatBtn.style.justifyContent = 'center';
    floatBtn.style.boxShadow = '0 2px 8px #00000030';
    floatBtn.style.cursor = 'grab';
    floatBtn.textContent = '📋';

    // 主面板
    const panel = document.createElement('div');
    panel.style.position = 'fixed';
    panel.style.zIndex = '999998';
    panel.style.background = '#fff';
    panel.style.border = '1px solid #ccc';
    panel.style.borderRadius = '8px';
    panel.style.padding = '12px';
    panel.style.width = '340px';
    panel.style.height = 'auto';
    panel.style.minWidth = '240px';
    panel.style.minHeight = '320px';
    panel.style.maxWidth = '90vw';
    panel.style.maxHeight = '80vh';
    panel.style.boxShadow = '0 2px 10px #00000022';
    panel.style.display = 'none';
    panel.style.resize = 'both';
    panel.style.overflow = 'auto';

    panel.innerHTML = `
        <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:8px;">
            <div style="font-weight:bold">网页内容复制</div>
            <div style="display:flex;gap:4px;align-items:center">
                <button id="drag_move_btn" style="padding:3px 6px;border:0;background:#666;color:#fff;border-radius:4px;cursor:grab;">✚</button>
                <button id="inbox_min_btn" style="padding:2px 6px;border:0;background:#ddd;border-radius:4px;cursor:pointer">✕</button>
            </div>
        </div>
        <div style="margin-bottom:6px">
            <label>标题（可选）</label>
            <input id="copy_title" style="width:100%;box-sizing:border-box;padding:4px" placeholder="标题">
        </div>
        <div style="margin-bottom:8px">
            <label>正文内容</label>
            <textarea id="copy_content" rows="5" style="width:100%;box-sizing:border-box;padding:4px" placeholder="正文..."></textarea>
        </div>
        <div style="display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px">
            <button id="tag_btn" style="padding:5px">#</button>
            <button id="todo_btn" style="padding:5px">待办</button>
            <button id="ul_btn" style="padding:5px">无序列表</button>
        </div>
        <div style="display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px">
            <button id="fill_page" style="flex:1;padding:5px">提取页面</button>
            <button id="clear_btn" style="flex:1;padding:5px;background:#999;color:#fff;border:0;border-radius:4px">清空</button>
            <button id="copy_btn" style="flex:1;padding:5px;background:#2277dd;color:#fff;border:0;border-radius:4px">一键复制</button>
            <button id="show_history" style="flex:1;padding:5px;background:#33aa55;color:#fff;border:0;border-radius:4px">历史</button>
        </div>
        <div id="msg_box" style="margin-top:4px;color:#666"></div>
        <div id="history_box" style="margin-top:8px;max-height:160px;overflow:auto;border:1px solid #ddd;padding:6px;display:none;font-size:12px">
            <div style="font-weight:bold;margin-bottom:4px">历史记录（点击填充）</div>
            <div id="history_list"></div>
        </div>
    `;

    document.body.appendChild(floatBtn);
    document.body.appendChild(panel);

    const $ = s => panel.querySelector(s);
    const titleInput = $('#copy_title');
    const contentInput = $('#copy_content');
    const msgBox = $('#msg_box');
    const historyBox = $('#history_box');
    const historyList = $('#history_list');
    const minBtn = $('#inbox_min_btn');
    const dragMoveBtn = $('#drag_move_btn');
    const copyBtn = $('#copy_btn');

    // ========== 十字按钮拖拽面板 ==========
    let panelDrag = {
        isDrag: false,
        startX: 0,
        startY: 0,
        panelLeft: 0,
        panelTop: 0
    };
    dragMoveBtn.addEventListener('mousedown', (e) => {
        e.preventDefault();
        panelDrag.isDrag = true;
        panelDrag.startX = e.clientX;
        panelDrag.startY = e.clientY;
        panelDrag.panelLeft = panel.offsetLeft;
        panelDrag.panelTop = panel.offsetTop;
        document.body.style.userSelect = 'none';
        dragMoveBtn.style.cursor = 'grabbing';
    })
    document.addEventListener('mousemove', (e) => {
        if (!panelDrag.isDrag) return;
        const dx = e.clientX - panelDrag.startX;
        const dy = e.clientY - panelDrag.startY;
        panel.style.left = (panelDrag.panelLeft + dx) + 'px';
        panel.style.top = (panelDrag.panelTop + dy) + 'px';
    })
    document.addEventListener('mouseup', () => {
        if (!panelDrag.isDrag) return;
        panelDrag.isDrag = false;
        document.body.style.userSelect = '';
        dragMoveBtn.style.cursor = 'grab';
    })

    // ========== 输入框聚焦自动上移避让输入法 ==========
    function checkInputFocus() {
        const activeEl = document.activeElement;
        if(activeEl === titleInput || activeEl === contentInput){
            const rect = panel.getBoundingClientRect();
            const winH = window.innerHeight;
            if(rect.bottom > winH - 80){
                let newTop = rect.top - INPUT_OFFSET;
                newTop = Math.max(10, newTop);
                panel.style.top = newTop + 'px';
            }
        }
    }
    titleInput.addEventListener('focus', checkInputFocus);
    contentInput.addEventListener('focus', checkInputFocus);

    // 光标插入文本
    function insertText(textarea, insertStr) {
        const start = textarea.selectionStart;
        const end = textarea.selectionEnd;
        const originVal = textarea.value;
        textarea.value = originVal.substring(0, start) + insertStr + originVal.substring(end);
        textarea.selectionStart = textarea.selectionEnd = start + insertStr.length;
        textarea.focus();
    }
    $('#tag_btn').onclick = () => insertText(contentInput, "#");
    $('#todo_btn').onclick = () => insertText(contentInput, "- [ ] ");
    $('#ul_btn').onclick = () => insertText(contentInput, "- ");

    // Ctrl+Enter复制
    contentInput.addEventListener('keydown', function(e) {
        if (e.ctrlKey && e.key === 'Enter') {
            e.preventDefault();
            copyBtn.click();
        }
    });

    // 悬浮球拖拽逻辑【修复点击冲突】
    let isDragging = false;
    let dragStartX, dragStartY, btnStartLeft, btnStartTop;
    let dragMoved = false;

    floatBtn.addEventListener('mousedown', startDrag);
    floatBtn.addEventListener('touchstart', startDrag, {passive:true});

    function startDrag(e) {
        isDragging = true;
        dragMoved = false;
        floatBtn.style.cursor = 'grabbing';
        const touch = e.touches ? e.touches[0] : e;
        dragStartX = touch.clientX;
        dragStartY = touch.clientY;
        const rect = floatBtn.getBoundingClientRect();
        btnStartLeft = rect.left;
        btnStartTop = rect.top;
        floatBtn.style.transform = 'none';
        document.addEventListener('mousemove', onDrag);
        document.addEventListener('touchmove', onDrag, {passive:true});
        document.addEventListener('mouseup', endDrag);
        document.addEventListener('touchend', endDrag);
    }

    function onDrag(e) {
        if (!isDragging) return;
        const touch = e.touches ? e.touches[0] : e;
        const dx = touch.clientX - dragStartX;
        const dy = touch.clientY - dragStartY;
        if(Math.abs(dx) > 6 || Math.abs(dy) >6) dragMoved = true;
        let newLeft = btnStartLeft + dx;
        let newTop = btnStartTop + dy;
        const w = floatBtn.offsetWidth;
        const h = floatBtn.offsetHeight;
        const winW = window.innerWidth;
        const winH = window.innerHeight;
        newLeft = Math.max(0, Math.min(newLeft, winW - w));
        newTop = Math.max(0, Math.min(newTop, winH - h));
        floatBtn.style.left = newLeft + 'px';
        floatBtn.style.top = newTop + 'px';
        floatBtn.style.right = 'auto';
        floatBtn.style.bottom = 'auto';
    }

    function endDrag() {
        if (!isDragging) return;
        isDragging = false;
        floatBtn.style.cursor = 'grab';
        document.removeEventListener('mousemove', onDrag);
        document.removeEventListener('touchmove', onDrag);
        document.removeEventListener('mouseup', endDrag);
        document.removeEventListener('touchend', endDrag);

        const rect = floatBtn.getBoundingClientRect();
        const centerX = rect.left + rect.width / 2;
        const winW = window.innerWidth;
        if (centerX > winW / 2) {
            floatBtn.style.right = '0px';
            floatBtn.style.left = 'auto';
        } else {
            floatBtn.style.left = '0px';
            floatBtn.style.right = 'auto';
        }
        floatBtn.style.top = rect.top + 'px';
        floatBtn.style.transform = 'translateY(0)';
    }

    function updatePanelPosition(){
        const rect = floatBtn.getBoundingClientRect();
        const winW = window.innerWidth;
        const winH = window.innerHeight;
        panel.style.display = 'block';
        panel.style.visibility = 'hidden';
        const panelW = panel.offsetWidth;
        const panelH = panel.offsetHeight;
        panel.style.visibility = 'visible';
        panel.style.display = 'none';

        let panelLeft = rect.left + rect.width + 8;
        let panelTop = rect.top;

        if(panelLeft + panelW > winW){
            panelLeft = rect.left - panelW -8;
        }
        let panelBottom = panelTop + panelH;
        if(panelBottom > winH){
            panelTop = winH - panelH - 10;
        }
        if(panelTop < 0){
            panelTop = 10;
        }
        panel.style.left = panelLeft + 'px';
        panel.style.top = panelTop + 'px';
    }

    // ========== 修复：使用click事件，增加延时判断拖拽状态 ==========
    floatBtn.addEventListener('click', () => {
        setTimeout(()=>{
            if(dragMoved) return;
            if (panel.style.display === 'none') {
                updatePanelPosition();
                panel.style.display = 'block';
                renderHistory();
            } else {
                collapsePanel();
            }
        },20)
    });

    function collapsePanel() {
        panel.style.display = 'none';
        historyBox.style.display = 'none';
    }

    // 历史记录
    function getHistory() {
        const raw = GM_getValue(HISTORY_KEY, '[]');
        try {
            return JSON.parse(raw);
        } catch {
            return [];
        }
    }
    function saveHistoryItem(title, content) {
        const list = getHistory();
        list.unshift({ title, content, time: new Date().toLocaleString() });
        const newList = list.slice(0, MAX_HISTORY);
        GM_setValue(HISTORY_KEY, JSON.stringify(newList));
        renderHistory();
    }
    function renderHistory() {
        const list = getHistory();
        historyList.innerHTML = "";
        if (list.length === 0) {
            historyList.innerHTML = "<div>暂无历史记录</div>";
            return;
        }
        list.forEach((item) => {
            const div = document.createElement('div');
            div.style.padding = '4px 0';
            div.style.borderBottom = '1px solid #eee';
            div.style.cursor = 'pointer';
            div.textContent = `【${item.time}】${item.title || '无标题'}`;
            div.onclick = () => {
                titleInput.value = item.title;
                contentInput.value = item.content;
                msgBox.textContent = "✅ 已载入选中历史";
                msgBox.style.color = 'green';
            };
            historyList.appendChild(div);
        });
    }

    // 提取页面标题链接
    function fillPageInfo() {
        const pageTitle = document.title;
        const pageUrl = window.location.href;
        const mdText = `[${pageTitle}](${pageUrl})\n\n`;
        contentInput.value += mdText;
        msgBox.textContent = "✅ 已提取页面标题和链接(MD)";
        msgBox.style.color = 'green';
    }
    $('#fill_page').onclick = fillPageInfo;

    // 清空
    $('#clear_btn').onclick = () => {
        titleInput.value = "";
        contentInput.value = "";
        msgBox.textContent = "✅ 已清空全部内容";
    };

    // 一键复制
    copyBtn.onclick = async () => {
        const titleVal = titleInput.value.trim();
        const contentVal = contentInput.value.trim();
        let copyText = "";
        if(titleVal) copyText += titleVal + "\n";
        copyText += contentVal;

        if(!copyText){
            msgBox.textContent = "❌ 内容为空，无法复制";
            msgBox.style.color = "red";
            return;
        }
        try{
            await GM_setClipboard(copyText);
            msgBox.textContent = "✅ 复制成功！";
            msgBox.style.color = 'green';
            saveHistoryItem(titleVal, contentVal);
        }catch(err){
            msgBox.textContent = "❌ 复制失败";
            msgBox.style.color = 'red';
        }
    };

    $('#show_history').onclick = () => {
        if(historyBox.style.display === 'none'){
            historyBox.style.display = 'block';
            renderHistory();
        }else{
            historyBox.style.display = 'none';
        }
    };

    minBtn.onclick = collapsePanel;

    // 划词自动拾取
    let selectionTimer = null;
    let lastSelectText = "";
    document.addEventListener('selectionchange', () => {
        clearTimeout(selectionTimer);
        selectionTimer = setTimeout(() => {
            const selText = window.getSelection().toString().trim();
            if (selText && selText !== lastSelectText) {
                lastSelectText = selText;
                contentInput.value += selText + "\n";
                updatePanelPosition();
                panel.style.display = 'block';
                msgBox.textContent = "✅ 已载入选中文字";
                msgBox.style.color = 'green';
            }
            if (!selText) {
                lastSelectText = "";
            }
        }, 350);
    });

})();
